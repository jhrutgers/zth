19/// Allow to yield execution from the current fiber to the scheduler.
20///
21/// When yielded too quickly, the fiber might just continue immediately.  It is safe to call this
22/// function often. When you cannot continue, and a yield is required, use [`out_of_work`].
23///
24/// This is a direct wrapper around the C API `zth_yield()`.
25pub fn may_yield() {
26    unsafe {
27        ffi::zth_yield();
28    }
29}
30
31/// Yields execution from the current fiber to the scheduler.
32///
33/// This is a direct wrapper around the C API `zth_outOfWork()`.
34pub fn out_of_work() {
35    unsafe {
36        ffi::zth_outOfWork();
37    }
38}

40/// Sleep the current fiber for at least the requested duration.
41///
42/// This function uses microsecond precision and chunks long sleeps to fit C `long`
43/// across platforms.
44pub fn sleep(duration: Duration) {
45    let mut sleep_for_us = duration.as_micros();
46    if sleep_for_us == 0 {
47        return;
48    }
49
50    let max_chunk = c_long::MAX as u128;
51    while sleep_for_us > 0 {
52        let chunk = core::cmp::min(sleep_for_us, max_chunk);
53        unsafe {
54            ffi::zth_unap(chunk as c_long);
55        }
56        sleep_for_us -= chunk;
57    }
58}
59
60/// Sleep the current fiber for the given number of milliseconds.
61pub fn sleep_ms(ms: u32) {
62    let max_chunk = c_long::MAX as u32;
63    let mut remaining = ms;
64
65    while remaining > 0 {
66        let chunk = core::cmp::min(remaining, max_chunk);
67        unsafe {
68            ffi::zth_mnap(chunk as c_long);
69        }
70        remaining -= chunk;
71    }
72}
73
74/// Sleep the current fiber for the given number of microseconds.
75pub fn sleep_us(us: u64) {
76    sleep(Duration::from_micros(us));
77}

79/// Fixed-delay periodic helper.
80///
81/// Each call to [`tick`](Self::tick) sleeps one full interval from the moment it is
82/// called. This is simple and works in `no_std`, but it does not compensate for work
83/// time drift.
84#[derive(Clone, Debug)]
85pub struct Period {
86    interval: Duration,
87    next_deadline: Option<Duration>,
88}
89
90impl Period {
91    /// Create a phase-locked periodic helper with a fixed interval.
92    pub const fn new(interval: Duration) -> Self {
93        Self {
94            interval,
95            next_deadline: None,
96        }
97    }
98
99    /// Returns the configured interval.
100    pub const fn interval(&self) -> Duration {
101        self.interval
102    }
103
104    /// Sleep until the next phase-locked tick.
105    ///
106    /// This keeps cadence aligned to the initial phase and compensates for variable
107    /// work time between calls. If monotonic time is unavailable, it falls back to
108    /// fixed-delay behavior.
109    pub fn tick(&mut self) {
110        if self.interval.is_zero() {
111            may_yield();
112            return;
113        }
114
115        let now = crate::time::now();
116
117        let deadline = match self.next_deadline {
118            Some(t) => t,
119            None => now.checked_add(self.interval).unwrap_or(Duration::MAX),
120        };
121
122        if deadline > now {
123            sleep(deadline - now);
124        }
125
126        let after = crate::time::now();
127        let mut next = deadline.checked_add(self.interval).unwrap_or(Duration::MAX);
128        while next <= after {
129            let Some(candidate) = next.checked_add(self.interval) else {
130                break;
131            };
132            next = candidate;
133        }
134
135        self.next_deadline = Some(next);
136    }
137}
